# Web framework
# Flask 3.1 + Werkzeug 3.1 close the remaining known advisories (the earlier
# 2.3.x pins carried ~15 CVEs between them). Verified to boot under the
# eventlet worker and pass the endpoint smoke tests.
flask==3.1.3
flask-socketio==5.6.1
flask-cors==6.0.5

# WebSocket support
python-socketio==5.17.0
eventlet==0.41.2

# Production server
# NOTE: gunicorn is intentionally kept on the 23.x line. gunicorn 26.x REMOVED
# the built-in eventlet worker (`--worker-class eventlet`), which is what
# render.yaml starts the app with. Upgrading gunicorn without first migrating
# off eventlet (e.g. to gevent or a threaded worker) would crash the web
# service on boot. gunicorn 23.0.0 has no known advisories.
gunicorn==23.0.0

# Authentication
pyjwt==2.15.1
werkzeug==3.1.9

# PostgreSQL (Neon)
psycopg2-binary==2.9.13

# HTTP requests (for Supabase Storage uploads)
requests==2.34.2

# Environment variables
python-dotenv==1.2.4
